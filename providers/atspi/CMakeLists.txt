cmake_minimum_required(VERSION 3.24)
project(RanvierWebCoreAtspi LANGUAGES C CXX)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
pkg_check_modules(ATSPI_HEADERS REQUIRED atspi-2)
if(NOT RANVIER_WEBCORE_LIBRARY)
    message(FATAL_ERROR "RANVIER_WEBCORE_LIBRARY must name the matching native host library")
endif()
add_library(RanvierWebCoreAtspi SHARED RanvierAtspi.cpp)
target_compile_features(RanvierWebCoreAtspi PRIVATE cxx_std_20)
target_include_directories(RanvierWebCoreAtspi PRIVATE ${ATSPI_HEADERS_INCLUDE_DIRS})
target_link_libraries(RanvierWebCoreAtspi PRIVATE PkgConfig::GIO "${RANVIER_WEBCORE_LIBRARY}")
set_target_properties(RanvierWebCoreAtspi PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
install(TARGETS RanvierWebCoreAtspi LIBRARY DESTINATION lib)
install(FILES RanvierAtspi.h DESTINATION include)
add_executable(atspi-layout ../../tests/atspi/layout.c)
enable_testing()
add_test(NAME atspi-layout COMMAND atspi-layout)
